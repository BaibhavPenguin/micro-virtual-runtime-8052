# Selected memory Map for System : 23-03-2026 : "uvr_variable.h"
	[Internal RAM Division]
		[30H - 7FH : Usable Internal Memory]

			[30H - 3FH : Page 1 Variables]
			[40H - 5FH : Program Buffer]
			[60H - 7FH : Hardware Stack]

			
		
# Decelared Page 1 Variables : 23-03-2026 : "uvr_variable.h"

	loop_counter at 30H		#Storing System wide loop variables
	
	prog_counter at 31H     	#Storing the current position of the program buffer
	
	rd_pointer at 32H       	#Read Pointer to read from  the program buffer
	
	wr_pointer at 33H       	#Write Buffer to modify the program buffer
	
	system_tick at 34H		#To Store System Time (Master Tick)
	
	op1 at 35H			#Store one of the operands
	
	op2 at 36H 			#Store another operand
	
	active_timer at 37H		#Store timer data for generating delays (system_tick - active_timer)

	instruction_buffer at 3DH 	#To Store Current Script Instruction
	
	result_l at 3EH			#To Store Lower Byte of Result
	
	result_h at 3FH			#To Store Higher Byte of Result

#Added Parser and Complete ByteCode execution sysrem 28-03-2026

	hash key = 00111111 (0x3F)
	Hashing Alhgorithm.
	The three bytes are byte1 , byte2 , byte3 respectively.
	instruction = byte1 XOR (byte2 AND hash key)
	instruction = instruction >> 1
	instruction = instruction ^ byte3
